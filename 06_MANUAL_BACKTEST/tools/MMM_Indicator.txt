// ⚠ THE SINGLE MOST IMPORTANT CAVEAT IN THIS FILE.
//
// The course prints a complete session table (see SECTION 3) and prints NO
// TIMEZONE ON IT. That is ambiguity record A-019 in
// 10_AMBIGUITIES/AUTOMATION_AMBIGUITIES.md, still OPEN, still `DO NOT CODE`.
// The instructor declines to specify (V02 [00:49:52] "Listen, don't analyse
// it… These are the times") and says the man who taught him has died
// (V02 [00:49:22]).
//
// DECISION D-031 converts the timezone from an assumption into a TESTED
// VARIABLE with two pre-registered arms, BOTH of which must always be reported:
//   ARM A — fixed UTC-5 year-round ("EST", no DST). The default.
//   ARM B — America/New_York, DST-tracking (UTC-4 in summer).
// Divergence between the arms is a FINDING, never a selection criterion.
// Reporting only the better-looking arm is error class E09/E24 (cherry-picking).
//
// This input exists so both arms can actually be run. Run both. Report both.
//
// CROSS-SOURCE OBSERVATION (recorded, NOT treated as closing A-019):
//   MMM-NOTES p.8 gives the same sessions in GMT with an ET gloss —
//   "Gap time between 07:00 and 07:30 GMT / 03:00 and 03:30 ET". 07:00 GMT
//   == 03:00 ET only at UTC-4, i.e. the Tier-2 document's own arithmetic is
//   DST-tracking (Arm B). Under Arm B the Tier-2 GMT table and the Tier-1
//   slide agree on Asian and London to the minute. They differ on the US close
//   (p.8 has 20:30 GMT = 16:30 ET; the Tier-1 slide has 5pm). This is
//   corroboration worth knowing and is NOT a resolution: it is one document's
//   internal arithmetic, and A-019 asks what the COURSE meant. A-019 stays open.

grpTz = "Session timezone  —  A-019 OPEN / D-031: run BOTH arms, report BOTH"

tzChoice = input.string("Arm A — UTC-5 (fixed EST, D-031 default)", "Timezone arm",
     options = ["Arm A — UTC-5 (fixed EST, D-031 default)",
                "Arm B — America/New_York (DST-tracking)",
                "UTC",
                "Chart / exchange timezone"],
     group = grpTz,
     tooltip = "A-019 is UNRESOLVED. D-031 requires both arms be pre-registered and both results reported. This is not a preference setting.")

    "Arm A — UTC-5 (fixed EST, D-031 default)" => "UTC-5"
    "Arm B — America/New_York (DST-tracking)"  => "America/New_York"

sesNyO   = input.session("0900-1000", "NY open session", group = gOpen,
     tooltip = "[TOOLING] sm_WorkTime Begin_5b=16:00/End_5b=17:00 server, converted at the INFERRED GMT+3. Brackets the 9:30 equities open MMM-NOTES p.8 Note 1 calls out.")
bgOpen   = input.color(color.new(#FFFFFF, 88), "Fill",   group = gOpen, inline = "co")
bcOpen   = input.color(color.new(#FFFFFF, 15), "Border", group = gOpen, inline = "co")

// -------------------------------------------------------------------------
// Box engine
// -------------------------------------------------------------------------
// Session boxes are only meaningful on intraday timeframes: on a daily or
// higher chart a single bar spans every session, so no box could be honest.
// Drawing is suppressed rather than approximated.
intradayOk = timeframe.isintraday

// Returns true when the current bar falls inside `sess` in timezone `tz`.
// Parameters are declared `simple string` deliberately: `time()` requires
// simple-qualified session and timezone arguments, and letting Pine infer a
// `series` qualifier here would fail to match the overload.
f_in(simple string sess, simple string days) =>
    not na(time(timeframe.period, sess + ":" + days, tz))

// One persistent box-set per call site. `var` inside a function gives each
// call site its own independent state, which is exactly what is wanted here.
// `isIn` / `wasIn` are passed in rather than derived from history inside the
// function body, so the function makes no history reference of its own.
f_box(bool isIn, bool wasIn, color bg, color bc, string txt, int keep, int extMs) =>
    var array<box> store = array.new<box>()
    var box   cur = na
    var float hi  = na
    var float lo  = na
    if isIn
        if not wasIn
            hi := high
            lo := low
            cur := box.new(
                 left        = time,
                 top         = hi,
                 bottom      = lo,
                 right       = time + extMs,
                 xloc        = xloc.bar_time,
                 bgcolor     = bg,
                 border_color= bc,
                 border_width= brdWidth,
                 text        = showLabels ? txt : "",
                 text_color  = bc,
                 text_size   = size.small,
                 text_halign = text.align_left,
                 text_valign = text.align_top)
            array.push(store, cur)
            if array.size(store) > keep
                box.delete(array.shift(store))
        else
            hi := math.max(hi, high)
            lo := math.min(lo, low)
            box.set_top(cur, hi)
            box.set_bottom(cur, lo)
            box.set_right(cur, time + extMs)

// Milliseconds per bar, for the optional right-extension.
extMs = extendBars * timeframe.in_seconds(timeframe.period) * 1000

draw = showBoxes and intradayOk

inAsia = draw and onAsia  and f_in(sesAsia, sessDays)
inLon  = draw and onLon   and f_in(sesLon,  sessDays)
inLonP = draw and onLonP  and f_in(sesLonP, sessDays)
inNy   = draw and onNy    and f_in(sesNy,   sessDays)
inNyP  = draw and onNyP   and f_in(sesNyP,  sessDays)
inGap  = draw and onGap   and f_in(sesGap,  sessDays)
inLonO = draw and onLonO  and f_in(sesLonO, sessDays)
inNyO  = draw and onNyO   and f_in(sesNyO,  sessDays)

// -------------------------------------------------------------------------
// Previous-bar session state.
//
// ⚠ THIS REPLACES A BUG. The earlier version wrote `nz(inAsia[1], false)`,
//    which does not compile:
//
//      Cannot call "nz" with argument "source"="call "operator SQBR"
//      (series bool)". An argument of "series bool" type was used but a
//      "simple int" is expected.  (CE10123)
//
//    `nz()` has overloads for the numeric and colour types but NOT for
//    `series bool`, so the compiler falls through to `nz(simple int, …)` and
//    rejects the boolean.
//
//    The fix carries the previous value in an explicit `var bool` instead of
//    reaching into history at all. That is also version-proof: it does not
//    depend on whether `bool` may hold `na` (v5) or may not (v6), and it needs
//    no na-guard in either language version.
// -------------------------------------------------------------------------
var bool wasAsia = false
var bool wasLon  = false
var bool wasLonP = false
var bool wasNy   = false
var bool wasNyP  = false
var bool wasGap  = false
var bool wasLonO = false
var bool wasNyO  = false

// Draw order: parent sessions first, then the nested boxes on top.
f_box(inAsia, wasAsia, bgAsia, bcAsia, txAsia, keepN, extMs)
f_box(inLon,  wasLon,  bgLon,  bcLon,  txLon,  keepN, extMs)
f_box(inNy,   wasNy,   bgNy,   bcNy,   txNy,   keepN, extMs)
f_box(inGap,  wasGap,  bgGap,  bcGap,  txGap,  keepN, extMs)
f_box(inLonP, wasLonP, bgLonP, bcLonP, txLonP, keepN, extMs)
f_box(inNyP,  wasNyP,  bgNyP,  bcNyP,  txNyP,  keepN, extMs)
f_box(inLonO, wasLonO, bgOpen, bcOpen, "LDN OPEN (tooling arm)", keepN, extMs)
f_box(inNyO,  wasNyO,  bgOpen, bcOpen, "NY OPEN (tooling arm)",  keepN, extMs)

// The 5pm High/Low Reset — a vertical line at the dead gap's opening bar.
// Same 17:00 boundary as `sesGap`, so the two cannot drift apart.
var array<line> rstLines = array.new<line>()
if inGap and not wasGap and showRst
    line ln = line.new(time, low, time, high,
         xloc      = xloc.bar_time,
         extend    = extend.both,
         color     = colRst,
         style     = line.style_dotted,
         width     = 1)
    array.push(rstLines, ln)
    if array.size(rstLines) > keepN
        line.delete(array.shift(rstLines))

// Roll the state forward only AFTER every box has consumed it.
wasAsia := inAsia
wasLon  := inLon
wasLonP := inLonP
wasNy   := inNy
wasNyP  := inNyP
wasGap  := inGap
wasLonO := inLonO
wasNyO  := inNyO

// A single, quiet warning when boxes are requested on a non-intraday chart,
// so the absence of boxes is never mistaken for "no sessions found".
// Held in a `var` and moved rather than re-created, so realtime bars do not
// accumulate one label per closed bar.
var label tfWarn = na
if showBoxes and not intradayOk and barstate.islast
    label.delete(tfWarn)
    tfWarn := label.new(bar_index, high, "Session boxes need an intraday timeframe",
         style     = label.style_label_down,
         color     = color.new(color.orange, 20),
         textcolor = color.white,
         size      = size.small)
