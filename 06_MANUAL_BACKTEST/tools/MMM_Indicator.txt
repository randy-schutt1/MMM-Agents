// -------------------------------------------------------------------------
// Box engine
// -------------------------------------------------------------------------
// Session boxes are only meaningful on intraday timeframes: on a daily or
// higher chart a single bar spans every session, so no box could be honest.
// Drawing is suppressed rather than approximated.
intradayOk = timeframe.isintraday

// Returns true when the current bar falls inside `sess` in timezone `tz`.
f_in(string sess, string days) =>
    not na(time(timeframe.period, sess + ":" + days, tz))

// One persistent box-set per call site. `var` inside a function gives each
// call site its own independent state, which is exactly what is wanted here.
// `isIn` / `wasIn` are passed in (rather than computed with [1] inside the
// function) so no history reference is made from within a function body.
f_box(bool isIn, bool wasIn, color bg, color bc, string txt, int keep, int extMs) =>
    var box[] store = array.new_box()
    var box   cur   = na
    var float hi    = na
    var float lo    = na
    if isIn
        if not wasIn
            hi := high
            lo := low
            cur := box.new(
                 left        = time,
                 top         = hi,
                 bottom      = lo,
                 right       = time + extMs,
                 xloc        = xloc.bar_time,
                 bgcolor     = bg,
                 border_color= bc,
                 border_width= brdWidth,
                 text        = showLabels ? txt : "",
                 text_color  = bc,
                 text_size   = size.small,
                 text_halign = text.align_left,
                 text_valign = text.align_top)
            array.push(store, cur)
            if array.size(store) > keep
                box.delete(array.shift(store))
        else
            hi := math.max(hi, high)
            lo := math.min(lo, low)
            box.set_top(cur, hi)
            box.set_bottom(cur, lo)
            box.set_right(cur, time + extMs)

// Milliseconds per bar, for the optional right-extension.
extMs = extendBars * timeframe.in_seconds(timeframe.period) * 1000

inNy   = draw and onNy    and f_in(sesNy,   sessDays)
inNyP  = draw and onNyP   and f_in(sesNyP,  sessDays)

// Draw order: parent sessions first, then the nested "prime" boxes on top.
f_box(inAsia, nz(inAsia[1], false), bgAsia, bcAsia, txAsia, keepN, extMs)
f_box(inLon,  nz(inLon[1],  false), bgLon,  bcLon,  txLon,  keepN, extMs)
f_box(inNy,   nz(inNy[1],   false), bgNy,   bcNy,   txNy,   keepN, extMs)
f_box(inLonP, nz(inLonP[1], false), bgLonP, bcLonP, txLonP, keepN, extMs)
