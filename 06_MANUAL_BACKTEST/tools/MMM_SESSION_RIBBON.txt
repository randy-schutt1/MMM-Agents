indicator("MMM Session Ribbon (!sm_WorkTime v1.5b replication)", overlay = true, max_boxes_count = 500, max_lines_count = 500)

// ── THE CLOCK — see WARNING 1. This is the load-bearing input. ───────────────
ribbonTZ = input.string("UTC+2", "Ribbon clock (MT4 server offset)", options = ["UTC-5","UTC-4","UTC+0","UTC+1","UTC+2","UTC+3"], tooltip = "The MT4 SERVER clock the saved times are in. UNVERIFIED — see WARNING 1. UTC+2/+3 is the standard broker clock and is a project INFERENCE, not a source.")
nDays    = input.int(50, "NumberOfDays", minval = 1, maxval = 500, tooltip = "MMM.tpl saves 50.")

// ── BAND 1 — ASIAN ──────────────────────────────────────────────────────────
gA = "Asian  (Begin_1 / End_1)"
drawAsian = input.bool(true,    "draw_asian_box",   group = gA)
aBeg      = input.string("00:00","Begin_1",         group = gA)
aEnd      = input.string("08:00","End_1",           group = gA)
aCol      = input.color(#8DDCE6, "Color_1",         group = gA, tooltip = "MT4 15128749 = BGR -> RGB #8DDCE6")
drawMid   = input.bool(true,    "draw_asian_midline", group = gA)
midCol    = input.color(#FFFF00,"asian_midline_color", group = gA, tooltip = "MT4 65535 = yellow")
midHours  = input.int(9,        "asian_midline_hours", group = gA, tooltip = "[GUESS] see MIDLINE note")

// ── BAND 2 — EURO (saved OFF) ───────────────────────────────────────────────
gE = "Euro  (Begin_2 / End_2)  -- saved OFF"
drawEuro = input.bool(false,    "draw_euro_box",    group = gE, tooltip = "MMM.tpl saves FALSE. Configured but not drawn.")
eBeg     = input.string("08:30","Begin_2",          group = gE)
eEnd     = input.string("14:00","End_2",            group = gE)
eCol     = input.color(#69690A, "Color_2",          group = gE)

// ── BAND 3 — NEW YORK ───────────────────────────────────────────────────────
gN = "New York  (Begin_3 / End_3)"
drawNY = input.bool(true,     "draw_ny_box",        group = gN)
nBeg   = input.string("16:30","Begin_3",            group = gN)
nEnd   = input.string("20:00","End_3",              group = gN)
nCol   = input.color(#9F7080, "Color_3",            group = gN)

// ── BAND 5 — MARKET OPEN (two disjoint windows) ─────────────────────────────
gM = "Market open  (Begin_5a/5b)"
drawMkt = input.bool(true,     "draw_mktopen_box",  group = gM)
m1Beg   = input.string("10:00","Begin_5a",          group = gM)
m1End   = input.string("11:00","End_5a",            group = gM)
m2Beg   = input.string("16:00","Begin_5b",          group = gM)
m2End   = input.string("17:00","End_5b",            group = gM)
mCol    = input.color(#B0B0DE, "Color_5",           group = gM)

// ── OPTIONAL: the 25/50 lines the MT4 build ships as ALERTS ─────────────────
// `Alert25BlueBox` / `Alert50BlueBox`, whose strings in the readable sibling
// build are literally "25 Pips above blue box" / "50 Pips below blue box".
// MMM.tpl saves BOTH false. They are drawn here only if switched on, and they
// are NOT a course rule -- see A-005, which is blocked on A-019/D-031.
gB = "Asian-box 25/50 pip bands  (saved OFF)"
show25 = input.bool(false, "Alert25BlueBox as lines", group = gB)
show50 = input.bool(false, "Alert50BlueBox as lines", group = gB)
bandCol= input.color(#808080, "band colour",         group = gB)

// ── helpers ─────────────────────────────────────────────────────────────────
// "HH:MM" -> minutes past midnight, in the ribbon clock.
mins(string hhmm) =>
    p = str.split(hhmm, ":")
    array.size(p) < 2 ? na : str.tonumber(array.get(p, 0)) * 60 + str.tonumber(array.get(p, 1))

// Bar's minutes-past-midnight ON THE RIBBON CLOCK.
nowMin = hour(time, ribbonTZ) * 60 + minute(time, ribbonTZ)

// [begin, end) membership. NOTE: none of the six saved windows wraps midnight,
// so wrap-around is NOT handled. If a future config wraps, this is WRONG.
inWin(string b, string e) =>
    mb = mins(b), me = mins(e)
    na(mb) or na(me) ? false : (nowMin >= mb and nowMin < me)

withinLookback = (last_bar_index - bar_index) < nDays * (1440 / math.max(timeframe.in_seconds() / 60, 1))

// ── § EXTREMES — [GUESS], and the reason it is flagged ──────────────────────
// The MT4 build's box top/bottom rule is NOT readable. This uses the standard
// MT4 session-box idiom, taken from the ONE readable session indicator on the
// volume -- KimIV's `Sessions.mq4`, which is a DIFFERENT and UNATTESTED tool:
//     p1 = High[Highest(...)]   p2 = Low[Lowest(...)]
// i.e. box top = session HIGH, box base = session LOW. That is the near-
// universal convention and it is still a GUESS about THIS build.
// ── a generic band tracker ──────────────────────────────────────────────────
type Band
    box   bx
    float hi
    float lo
    int   t0

var Band asian = Band.new(na, na, na, na)
var Band euro  = Band.new(na, na, na, na)
var Band ny    = Band.new(na, na, na, na)
var Band mk1   = Band.new(na, na, na, na)
var Band mk2   = Band.new(na, na, na, na)

step(Band s, bool active, color c, bool enabled) =>
    if enabled and withinLookback
        if active
            if na(s.bx)
                s.hi := high
                s.lo := low
                s.t0 := time
                s.bx := box.new(time, s.hi, time, s.lo, xloc = xloc.bar_time, border_color = c, bgcolor = color.new(c, 80))
            else
                s.hi := math.max(s.hi, high)
                s.lo := math.min(s.lo, low)
                box.set_top(s.bx,    s.hi)
                box.set_bottom(s.bx, s.lo)
                box.set_right(s.bx,  time)
        else
            s.bx := na
    s

asian := step(asian, inWin(aBeg,  aEnd),  aCol, drawAsian)
euro  := step(euro,  inWin(eBeg,  eEnd),  eCol, drawEuro)
ny    := step(ny,    inWin(nBeg,  nEnd),  nCol, drawNY)
mk1   := step(mk1,   inWin(m1Beg, m1End), mCol, drawMkt)
mk2   := step(mk2,   inWin(m2Beg, m2End), mCol, drawMkt)

// ── § MIDLINE — [GUESS], and it is the weaker of the two ────────────────────
// `draw_asian_midline=true`, `asian_midline_color=65535` (yellow),
// `asian_midline_hours=9`.
// ⚠️ THE SAVED ASIAN BOX IS 8 HOURS LONG (0:00-8:00) AND `midline_hours` IS 9.
//    A "midline" parameterised by a number LARGER than the box it belongs to
//    cannot mean "the middle of the box". The two readings that survive:
//      (a) the box mid-price, PROJECTED FORWARD 9 hours past End_1  <- used here
//      (b) a line drawn AT hour 9, unrelated to the box mid-price
//    NEITHER IS READABLE. (a) is chosen because it is the only one for which
//    "midline" and "hours" are both doing work. THIS IS A RECONSTRUCTION.
//    If the owner's chart shows otherwise, (a) is wrong and this is the first
//    thing the D-055 §4.3 acceptance test should look at.
var line midLn = na
asianEnded = drawMid and drawAsian and not inWin(aBeg, aEnd) and inWin(aBeg, aEnd)[1]
if asianEnded and not na(asian.hi) and not na(asian.lo)
    mid = (asian.hi + asian.lo) / 2
    midLn := line.new(time, mid, time + midHours * 3600 * 1000, mid, xloc = xloc.bar_time, color = midCol, width = 1)

// ── the 25/50 bands off the asian box, if enabled ───────────────────────────
pip = syminfo.mintick * (syminfo.type == "forex" and syminfo.mintick < 0.001 ? 10 : 1)
if asianEnded and not na(asian.hi)
    if show25
        line.new(time, asian.hi + 25 * pip, time + 12 * 3600 * 1000, asian.hi + 25 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dotted)
        line.new(time, asian.lo - 25 * pip, time + 12 * 3600 * 1000, asian.lo - 25 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dotted)
    if show50
        line.new(time, asian.hi + 50 * pip, time + 12 * 3600 * 1000, asian.hi + 50 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dashed)
        line.new(time, asian.lo - 50 * pip, time + 12 * 3600 * 1000, asian.lo - 50 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dashed)
