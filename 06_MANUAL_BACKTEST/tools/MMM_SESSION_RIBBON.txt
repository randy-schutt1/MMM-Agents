// ── THE SAVED SERVER-TIME BOUNDARIES (verbatim from MMM.tpl) ────────────────

aCol      = input.color(#8DDCE6, "Color_1",             group = gA, tooltip = "MT4 15128749 (BGR) -> #8DDCE6")

// ── helpers ─────────────────────────────────────────────────────────────────
// "HH:MM" (server) -> minutes past midnight IN THE RENDER CLOCK.
// This is where the owner's EST ruling is actually applied.
minsEST(string hhmm) =>
    p = str.split(hhmm, ":")
    array.size(p) < 2 ? na : math.round((str.tonumber(array.get(p, 0)) + shift) * 60 + str.tonumber(array.get(p, 1))) % 1440

// Bar's minutes past midnight on the render clock.
nowMin = hour(time, tz) * 60 + minute(time, tz)

// [begin, end) membership. ⭐ WRAP-AROUND IS HANDLED AND IS NOW REQUIRED:
// converted to EST the Asian band becomes 17:00-01:00, i.e. it CROSSES MIDNIGHT.
// The server-time version never did. This is a real consequence of the ruling.
inWin(string b, string e) =>
    mb = minsEST(b), me = minsEST(e)
    na(mb) or na(me) ? false : mb <= me ? (nowMin >= mb and nowMin < me) : (nowMin >= mb or nowMin < me)

barsPerDay     = 1440 / math.max(timeframe.in_seconds() / 60, 1)
withinLookback = (last_bar_index - bar_index) < nDays * barsPerDay

// ── § EXTREMES — [GUESS] ────────────────────────────────────────────────────
// The MT4 build's box top/bottom rule is NOT readable. This uses the standard
// MT4 session-box idiom, taken from the one readable session indicator on the
// volume (KimIV's Sessions.mq4 — a DIFFERENT, UNATTESTED tool, cited for
// mechanism only): box top = session HIGH, box base = session LOW.
type Band
    box   bx
    float hi
    float lo

var Band asian = Band.new(na, na, na)
var Band euro  = Band.new(na, na, na)
var Band ny    = Band.new(na, na, na)
var Band mk1   = Band.new(na, na, na)
var Band mk2   = Band.new(na, na, na)

step(Band s, bool active, color c, bool enabled) =>
    if enabled and withinLookback
        if active
            if na(s.bx)
                s.hi := high
                s.lo := low
                s.bx := box.new(time, s.hi, time, s.lo, xloc = xloc.bar_time, border_color = c, bgcolor = color.new(c, 80))
            else
                s.hi := math.max(s.hi, high)
                s.lo := math.min(s.lo, low)
                box.set_top(s.bx,    s.hi)
                box.set_bottom(s.bx, s.lo)
                box.set_right(s.bx,  time)
        else
            s.bx := na
    s

asian := step(asian, inWin(aBeg,  aEnd),  aCol, drawAsian)
euro  := step(euro,  inWin(eBeg,  eEnd),  eCol, drawEuro)
ny    := step(ny,    inWin(nBeg,  nEnd),  nCol, drawNY)
mk1   := step(mk1,   inWin(m1Beg, m1End), mCol, drawMkt)
mk2   := step(mk2,   inWin(m2Beg, m2End), mCol, drawMkt)

//      (a) the box mid-price PROJECTED FORWARD 9 hours past End_1  <- used here

    mid = (asian.hi + asian.lo) / 2
    line.new(time, mid, time + midHours * 3600 * 1000, mid, xloc = xloc.bar_time, color = midCol, width = 1)

// ── the 25/50 bands off the asian box, if enabled ───────────────────────────

        line.new(time, asian.lo - 25 * pip, time + 12 * 3600 * 1000, asian.lo - 25 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dotted)
    if show50
        line.new(time, asian.hi + 50 * pip, time + 12 * 3600 * 1000, asian.hi + 50 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dashed)
        line.new(time, asian.lo - 50 * pip, time + 12 * 3600 * 1000, asian.lo - 50 * pip, xloc = xloc.bar_time, color = bandCol, style = line.style_dashed)
