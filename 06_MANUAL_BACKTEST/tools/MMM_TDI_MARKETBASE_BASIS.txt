indicator(
     title      = "MMM — TDI  [VARIANT B: bands on MARKET BASE]  (V14 45:09 · C-021 DISPUTED)",
     shorttitle = "MMM TDI · B · MarketBase-basis",
     overlay    = false,
     precision  = 2)

// -------------------------------------------------------------------------
// Core parameters
// -------------------------------------------------------------------------
grpP = "TDI parameters  —  RSI 21 is [TIER 1]; the rest [TOOLING]"

src      = input.source(close, "RSI source", group = grpP,
     tooltip = "[TOOLING] !SM_TDI has RSI_Price=0, which is MT4 PRICE_CLOSE.")
rsiLen   = input.int(21,  "RSI period",                minval = 1, group = grpP,
     tooltip = "[TIER 1] A-080 is CLOSED, RESOLVED BY COURSE: the RSI period is 21. NOT the 13 that circulates publicly — that is a Tier-3 default and it is wrong for this method.")
fastLen  = input.int(2,   "Fast MA (RSI Price Line)",  minval = 1, group = grpP,
     tooltip = "[TOOLING] !SM_TDI RSI_Price_Line=2, RSI_Price_Type=0 (MT4 MODE_SMA).")
slowLen  = input.int(7,   "Slow MA (Trade Signal Line)", minval = 1, group = grpP,
     tooltip = "[TOOLING] !SM_TDI Trade_Signal_Line=7, Trade_Signal_Type=0 (MT4 MODE_SMA).")
bandLen  = input.int(34,  "Volatility band / Market Base Line period", minval = 2, group = grpP,
     tooltip = "[TOOLING] !SM_TDI Volatility_Band=34. ⚠️ A-086 — the bands' period is NEVER stated in Tier 1 or Tier 2. This is the TOOLING candidate for it, and A-086 stays ELIGIBLE, NOT CLOSED.")
bandMult = input.float(2.0, "Volatility band std-dev multiple", minval = 0.1, step = 0.1, group = grpP,
     tooltip = "[TIER 1] NOT DISPUTED. V12 and V14 both say 'two standard deviations'; V14 states it unhedged and drops V12's alternative '2%'. C-021 treats the multiplier as a RESOLUTION, not a conflict. Held identical across both variants so the basis is the only variable.")

maType = input.string("SMA", "MA type for the two signal lines", options = ["SMA", "EMA", "WMA"], group = grpP,
     tooltip = "[TOOLING] !SM_TDI RSI_Price_Type=0 and Trade_Signal_Type=0; MT4 mode 0 is SMA.")

f_ma(series float s, simple int len, simple string t) =>
    switch t
        "EMA" => ta.ema(s, len)
        "WMA" => ta.wma(s, len)
        => ta.sma(s, len)

// -------------------------------------------------------------------------
// Lines — colours [TOOLING] (MT4 BGR integers decoded from the template);
// buffer->line mapping INFERRED, not proven. Recolour freely.
// -------------------------------------------------------------------------
grpV = "Lines  —  [TOOLING] colours; buffer→line mapping INFERRED"

showRsi   = input.bool(false, "Raw RSI line",           group = grpV, inline = "v1",
     tooltip = "Off by default: !SM_TDI does not plot the bare RSI, it plots the Fast MA of it. ⭐ TURN THIS ON when comparing the two variants — in THIS file the bands are a deviation of the MARKET BASE LINE, not of this series, and seeing both makes the difference obvious.")
colRsi    = input.color(color.new(#B2B5BE, 40), "",     group = grpV, inline = "v1")
showFast  = input.bool(true,  "Fast MA / RSI Price Line", group = grpV, inline = "v2")
colFast   = input.color(#B0C4DE, "",                    group = grpV, inline = "v2")
showSlow  = input.bool(true,  "Slow MA / Trade Signal", group = grpV, inline = "v3")
colSlow   = input.color(#B22222, "",                    group = grpV, inline = "v3")
showBase  = input.bool(true,  "Market Base Line",       group = grpV, inline = "v4")
colBase   = input.color(#191970, "",                    group = grpV, inline = "v4")
showBands = input.bool(true,  "Volatility Bands",       group = grpV, inline = "v5")
colBands  = input.color(#1E90FF, "",                    group = grpV, inline = "v5")
fillBands = input.bool(true,  "Shade between the bands", group = grpV)

// -------------------------------------------------------------------------
// Reference levels — [TOOLING], template levels 68 / 63 / 50 / 37 / 32
// -------------------------------------------------------------------------
grpL = "Reference levels  —  [TOOLING] template levels 68 / 63 / 50 / 37 / 32"

showLevels  = input.bool(true, "Show 68 / 50 / 32", group = grpL)
lvlUp       = input.int(68, "Upper", group = grpL, inline = "lv")
lvlMid      = input.int(50, "Mid",   group = grpL, inline = "lv")
lvlDn       = input.int(32, "Lower", group = grpL, inline = "lv")

showShark   = input.bool(true, "Show shark-fin levels (63 / 37)", group = grpL,
     tooltip = "[TOOLING] !SM_TDI SharkFin_Upper_Level=63, SharkFin_Lower_Level=37. A-032 remains OPEN — these are the indicator's thresholds, not a definition of the pattern.")
sharkUp     = input.int(63, "Shark upper", group = grpL, inline = "sf")
sharkDn     = input.int(37, "Shark lower", group = grpL, inline = "sf")
colShark    = input.color(color.new(#FF9800, 35), "", group = grpL, inline = "sf")

// =========================================================================
// CALCULATION
//
// ⭐⭐ THE ONE LINE THAT DIFFERS BETWEEN THE TWO VARIANTS IS `dev`, BELOW. ⭐⭐
//
//   VARIANT A (other file): dev = bandMult * ta.stdev(rsiVal,   bandLen)
//   VARIANT B (this file):  dev = bandMult * ta.stdev(baseLine, bandLen)
//
// Everything above and below this block is identical in both files.
// =========================================================================
rsiVal = ta.rsi(src, rsiLen)

baseLine = ta.sma(rsiVal, bandLen)                  // Market Base Line — the band CENTRE in both variants

// ⭐ VARIANT B — bands are N std-devs OF THE MARKET BASE LINE.
//    V14 [00:45:09]: "The bands are two standard deviations away from the
//    market base."  Tier 2, MMM-NOTES p.45, independently agrees.
//    ⛔ DISPUTED — C-021 / item 187 / D-048 rung 4. See header.
//
//    NOTE ON WHAT THIS MEANS ARITHMETICALLY: `baseLine` is already a 34-period
//    SMA of the RSI, so taking a 34-period standard deviation OF it measures
//    the dispersion of an already-smoothed series. It is structurally a much
//    quieter number than variant A's, and that is the whole visual difference.
dev      = bandMult * ta.stdev(baseLine, bandLen)

bandUp   = baseLine + dev                           // upper Volatility Band
bandDn   = baseLine - dev                           // lower Volatility Band

fastMa = f_ma(rsiVal, fastLen, maType)              // RSI Price Line
slowMa = f_ma(rsiVal, slowLen, maType)              // Trade Signal Line

// -------------------------------------------------------------------------
// Plots
// -------------------------------------------------------------------------
pUp = plot(showBands ? bandUp : na, "Volatility Band Up",   color = colBands, linewidth = 1)
pDn = plot(showBands ? bandDn : na, "Volatility Band Down", color = colBands, linewidth = 1)
fill(pUp, pDn, color = fillBands and showBands ? color.new(colBands, 92) : na, title = "Band fill")

plot(showBase ? baseLine : na, "Market Base Line",          color = colBase, linewidth = 2)
plot(showRsi  ? rsiVal   : na, "RSI (raw)",                 color = colRsi,  linewidth = 1)
plot(showFast ? fastMa   : na, "Fast MA (RSI Price Line)",  color = colFast, linewidth = 2)
plot(showSlow ? slowMa   : na, "Slow MA (Trade Signal Line)", color = colSlow, linewidth = 2)

// Reference levels are drawn with `plot`, not `hline`: Pine requires `hline`'s
// price to be a const/input, so it cannot take a `showLevels ? … : na`
// expression. `plot` accepts the series and keeps the toggle working.
plot(showLevels ? lvlUp  : na, "Level 68", color = color.new(color.gray, 50), style = plot.style_linebr)
plot(showLevels ? lvlMid : na, "Level 50", color = color.new(color.gray, 20), style = plot.style_linebr)
plot(showLevels ? lvlDn  : na, "Level 32", color = color.new(color.gray, 50), style = plot.style_linebr)
plot(showShark ? sharkUp : na, "Shark fin upper (63)", color = colShark, style = plot.style_linebr)
plot(showShark ? sharkDn : na, "Shark fin lower (37)", color = colShark, style = plot.style_linebr)

         style     = label.style_label_down,
